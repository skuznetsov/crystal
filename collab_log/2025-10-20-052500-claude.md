### 2. Spec Structure for def/class Collection

require "spec"
require "./ast_fixtures"

require "../../src/compiler/frontend/ast"
require "../../src/compiler/semantic/context"
require "../../src/compiler/semantic/collectors/symbol_collector"
require "../../src/compiler/semantic/symbol"

alias Frontend = CrystalV2::Compiler::Frontend
alias Semantic = CrystalV2::Compiler::Semantic

describe Semantic::SymbolCollector do
  describe "def collection" do
    it "collects method definitions into the global symbol table" do
      arena = Frontend::AstArena.new
      def_id = AstFixtures.make_def(arena, "calculate", params: ["x", "y"])

      program = Frontend::Program.new(arena, [def_id])
      context = Semantic::Context.new(Semantic::SymbolTable.new)

      collector = Semantic::SymbolCollector.new(program, context)
      collector.collect

      symbol = context.symbol_table.lookup("calculate")
      symbol.should_not be_nil
      symbol.should be_a(Semantic::MethodSymbol)

      method_sym = symbol.as(Semantic::MethodSymbol)
      method_sym.params.should eq(["x", "y"])
    end

    it "allows method redefinition (Crystal semantics)" do
      arena = Frontend::AstArena.new
      first_def = AstFixtures.make_def(arena, "foo", params: ["a"])
      second_def = AstFixtures.make_def(arena, "foo", params: ["a", "b"])

      program = Frontend::Program.new(arena, [first_def, second_def])
      context = Semantic::Context.new(Semantic::SymbolTable.new)

      collector = Semantic::SymbolCollector.new(program, context)
      collector.collect

      symbol = context.symbol_table.lookup("foo")
      symbol.should_not be_nil
      method_sym = symbol.as(Semantic::MethodSymbol)
      method_sym.params.size.should eq(2)  # Last definition wins
    end
  end

  describe "class collection" do
    it "collects class definitions into the global symbol table" do
      arena = Frontend::AstArena.new
      class_id = AstFixtures.make_class(arena, "Person")

      program = Frontend::Program.new(arena, [class_id])
      context = Semantic::Context.new(Semantic::SymbolTable.new)

      collector = Semantic::SymbolCollector.new(program, context)
      collector.collect

      symbol = context.symbol_table.lookup("Person")
      symbol.should_not be_nil
      symbol.should be_a(Semantic::ClassSymbol)
    end

    it "creates nested scope for class body" do
      arena = Frontend::AstArena.new

      # Create method inside class
      method_id = AstFixtures.make_def(arena, "greet")
      class_id = AstFixtures.make_class(arena, "Person", body_ids: [method_id])

      program = Frontend::Program.new(arena, [class_id])
      context = Semantic::Context.new(Semantic::SymbolTable.new)

      collector = Semantic::SymbolCollector.new(program, context)
      collector.collect

      # Class should exist in global scope
      class_sym = context.symbol_table.lookup("Person")
      class_sym.should_not be_nil
      class_sym.should be_a(Semantic::ClassSymbol)

      # Method should exist in class's nested scope
      # Assuming ClassSymbol has a `scope` or `symbol_table` field
      class_scope = class_sym.as(Semantic::ClassSymbol).scope
      method_sym = class_scope.lookup("greet")
      method_sym.should_not be_nil
      method_sym.should be_a(Semantic::MethodSymbol)
    end

    it "supports class reopening (Crystal semantics)" do
      arena = Frontend::AstArena.new

      # First class definition with method `foo`
      method1_id = AstFixtures.make_def(arena, "foo")
      class1_id = AstFixtures.make_class(arena, "MyClass", body_ids: [method1_id])

      # Second class definition (reopening) with method `bar`
      method2_id = AstFixtures.make_def(arena, "bar")
      class2_id = AstFixtures.make_class(arena, "MyClass", body_ids: [method2_id])

      program = Frontend::Program.new(arena, [class1_id, class2_id])
      context = Semantic::Context.new(Semantic::SymbolTable.new)

      collector = Semantic::SymbolCollector.new(program, context)
      collector.collect

      class_sym = context.symbol_table.lookup("MyClass")
      class_sym.should_not be_nil

      # Both methods should exist in the class scope
      class_scope = class_sym.as(Semantic::ClassSymbol).scope
      class_scope.lookup("foo").should_not be_nil
      class_scope.lookup("bar").should_not be_nil
    end
  end
end

### 4. Symbol Hierarchy Extension Plan

**Recommended MethodSymbol structure**:

```crystal
# src/compiler/semantic/symbol.cr

class MethodSymbol < Symbol
  getter params : Array(String)
  getter return_type : TypeId?  # Optional, for annotated returns
  getter body : ExprId?          # nil for abstract methods

  def initialize(name : String, node_id : ExprId, @params : Array(String), @body : ExprId? = nil, @return_type : TypeId? = nil)
    super(name, node_id)
  end
end

class ClassSymbol < Symbol
  getter scope : SymbolTable         # Nested scope for members
  getter superclass : String?        # Name of superclass
  getter instance_vars : Hash(String, TypeId)

  def initialize(name : String, node_id : ExprId, @superclass : String? = nil)
    super(name, node_id)
    @scope = SymbolTable.new  # No parent - class scope is isolated
    @instance_vars = {} of String => TypeId
  end
end

class VariableSymbol < Symbol
  getter declared_type : TypeId?
  getter inferred_type : TypeId?

  def initialize(name : String, node_id : ExprId, @declared_type : TypeId? = nil)
    super(name, node_id)
    @inferred_type = nil
  end
end
```

**Key design notes**:
1. **ClassSymbol.scope**: Each class gets its own SymbolTable for methods/instance vars
2. **MethodSymbol.params**: Array of parameter names (types come in Stage 3)
3. **VariableSymbol**: For local variables and parameters

class NameResolver
  alias ResolutionMap = Hash(ExprId, Symbol)

  getter resolution_map : ResolutionMap
  getter diagnostics : Array(Diagnostic)

  def initialize(@program : Frontend::Program, @context : Context)
    @resolution_map = {} of ExprId => Symbol
    @diagnostics = [] of Diagnostic
    @current_scope = @context.symbol_table
  end

  def resolve
    @program.roots.each do |root_id|
      visit(root_id)
    end
    @diagnostics
  end

  private def visit(node_id : ExprId)
    node = @program.arena[node_id]

    case node.kind
    when .identifier?
      resolve_identifier(node_id, node)
    when .assign?
      # Visit RHS first, then define LHS
      visit(node.right.not_nil!) if node.right
      define_variable(node)
    when .class_def?
      # Push class scope, visit body, pop scope
      visit_class(node_id, node)
    when .def?
      # Push method scope, visit body, pop scope
      visit_method(node_id, node)
    # ... other cases
    end
  end

  private def resolve_identifier(node_id : ExprId, node : ExpressionNode)
    name = String.new(node.identifier_name.not_nil!)

    if symbol = @current_scope.lookup(name)
      @resolution_map[node_id] = symbol
    else
      @diagnostics << Diagnostic.new(
        "undefined local variable or method '#{name}'",
        node.span
      )
    end
  end

  private def visit_class(node_id : ExprId, node : ExpressionNode)
    name = String.new(node.class_name.not_nil!)
