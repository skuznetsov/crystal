```crystal

```

**Impact**:
- Diagnostics for duplicate/shadowed parameters point to entire `def` line
- Can't highlight specific parameter in error message
- E2002/W2001 less precise for parameters

### Proposed Solution

**Phase 1: Token-level parameter tracking** (minimal parser changes)

1. **Lexer enhancement**: Already tracks all token spans ✅

2. **Parser change** (src/compiler/frontend/parser.cr:parse_def):
   ```crystal

   params = [] of String

   # Proposed:
   params = [] of {String, Frontend::Span}  # (name, span)
   ```

3. **AST change** (src/compiler/frontend/ast.cr):
   ```crystal
   # Add new field to ExpressionNode:
   property def_param_spans : Array(Frontend::Span)?
   ```

4. **SymbolCollector usage**:
   ```crystal
   # Access param spans:
   param_spans = node.def_param_spans || []
   params.zip(param_spans) do |param_name, param_span|
     param_node_id = create_synthetic_node_for_param(param_span)
     param_symbol = VariableSymbol.new(param_name, param_node_id)
   ```

**Proposed changes** (src/compiler/cli.cr):

```crystal
private def compile_file(path, options, out_io, err_io)
  source = File.read(path)
  lexer = Frontend::Lexer.new(source)
  parser = Frontend::Parser.new(lexer)
  program = parser.parse_program

  if parser.diagnostics.empty?
    analyzer = Semantic::Analyzer.new(program)
    analyzer.collect_symbols

    if options.check_semantics
      result = analyzer.resolve_names
      report_collector_diagnostics(analyzer.semantic_diagnostics, source, err_io)
      report_resolution_diagnostics(result.diagnostics, source, err_io)

      # NEW: Check for semantic errors and exit
      if analyzer.semantic_errors?
        err_io.puts "\nerror: compilation failed due to semantic errors"
        exit 1
      end

      # NEW: Check for name resolution errors
      if result.diagnostics.any? { |d| d.message.starts_with?("undefined") }
        err_io.puts "\nerror: compilation failed due to name resolution errors"
        exit 1
      end
    end

    out_io.puts "Parsed #{program.roots.size} top-level expressions"
    if options.dump_symbols
      dump_symbols(program, analyzer.global_context.symbol_table, out_io)
    end
  else
    parser.diagnostics.each do |diag|
      err_io.puts Frontend::DiagnosticFormatter.format(source, diag)
    end
    # NEW: Parser errors should also fail compilation
    if options.check_semantics
      err_io.puts "\nerror: compilation failed due to parser errors"
      exit 1
    end
  end
rescue File::NotFoundError
  err_io.puts "error: file not found #{path}"
  exit 1
end
```

## Task 5: ⏳ Prepare for Stage 3 (Sketch)

**Request**: Design method overload sets and class symbol metadata for type inference

### Method Overload Sets

**Current limitation**: Only one MethodSymbol per name
```crystal
def foo(x : Int32); end
def foo(x : String); end  # Overwrites first definition
```

**Proposed design**:

```crystal
# New data structure:
class MethodOverloadSet
  getter name : String
  getter overloads : Array(MethodSymbol)

  def initialize(@name : String)
    @overloads = [] of MethodSymbol
  end

  def add_overload(method : MethodSymbol)
    # Check for exact duplicate signature
    if @overloads.any? { |m| same_signature?(m, method) }
      # Emit E2004: duplicate method signature
    else
      @overloads << method
    end
  end

  private def same_signature?(a : MethodSymbol, b : MethodSymbol) : Bool
    return false if a.params.size != b.params.size
    # Future: Compare parameter types, restrictions, etc.
    true
  end
end

# Symbol table modification:
class SymbolTable
  @symbols = {} of String => (Symbol | MethodOverloadSet)

  def define_method(name : String, method : MethodSymbol)
    existing = @symbols[name]?
    case existing
    when MethodSymbol
      # Convert to overload set
      set = MethodOverloadSet.new(name)
      set.add_overload(existing)
      set.add_overload(method)
      @symbols[name] = set
    when MethodOverloadSet
      set.add_overload(method)
    when nil
      @symbols[name] = method
    else
      # Error: can't overload non-method
    end
  end
end
```

```crystal
  def initialize(

    node_id : ExprId,
    *,
    scope : SymbolTable,

    type_params : Array(String)? = nil
```
