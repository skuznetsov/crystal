## Available Helpers

### Core Helpers

**make_def** - Create method definitions
```crystal
# Simple method:
def_id = AstFixtures.make_def(arena, "greet")

# With parameters:
def_id = AstFixtures.make_def(arena, "greet", params: ["name"])

# With body:
body_id = AstFixtures.make_identifier(arena, "name")
def_id = AstFixtures.make_def(arena, "greet", params: ["name"], body: [body_id])
```

**make_class** - Create class definitions
```crystal
# Empty class:
class_id = AstFixtures.make_class(arena, "Person")

# With methods:
method_id = AstFixtures.make_def(arena, "greet")
class_id = AstFixtures.make_class(arena, "Person", body: [method_id])

# With superclass (for future):
class_id = AstFixtures.make_class(arena, "Student", superclass: "Person")
```

**make_identifier** - Create identifier references
```crystal
id = AstFixtures.make_identifier(arena, "x")
```

**make_call** - Create method calls
```crystal
# Simple call:
call_id = AstFixtures.make_call(arena, "greet")

# With arguments:
arg_id = AstFixtures.make_string(arena, "World")
call_id = AstFixtures.make_call(arena, "greet", args: [arg_id])
```

**make_number** - Create number literals
```crystal
num_id = AstFixtures.make_number(arena, 42)
```

**make_string** - Create string literals
```crystal
str_id = AstFixtures.make_string(arena, "hello")
```

**make_macro** - Create macro definitions
```crystal
macro_id = AstFixtures.make_macro(arena, "greet")
```

## Usage Examples

### Example 1: Simple Method
```crystal
arena = Frontend::AstArena.new

# Create: def greet(name)
#           name
#         end
name_ref = AstFixtures.make_identifier(arena, "name")
def_id = AstFixtures.make_def(arena, "greet", params: ["name"], body: [name_ref])

# Verify
node = arena[def_id]
node.kind.should eq(ExpressionNode::Kind::Def)
String.new(node.def_name.not_nil!).should eq("greet")
node.def_params.should eq(["name"])
```

### Example 2: Class with Method
```crystal
arena = Frontend::AstArena.new

# Create: class Greeter
#           def say_hello
#             greet("World")
#           end
#         end
arg = AstFixtures.make_string(arena, "World")
call = AstFixtures.make_call(arena, "greet", args: [arg])
method = AstFixtures.make_def(arena, "say_hello", body: [call])
klass = AstFixtures.make_class(arena, "Greeter", body: [method])

# Verify
class_node = arena[klass]
class_node.kind.should eq(ExpressionNode::Kind::Class)
class_node.class_body.not_nil!.size.should eq(1)
```

### Example 3: Multiple Methods in Class
```crystal
arena = Frontend::AstArena.new

# Create: class Person
#           def greet
#           end
#           def goodbye
#           end
#         end
greet = AstFixtures.make_def(arena, "greet")
goodbye = AstFixtures.make_def(arena, "goodbye")
klass = AstFixtures.make_class(arena, "Person", body: [greet, goodbye])

class_node = arena[klass]
class_node.class_body.not_nil!.size.should eq(2)
```

## Integration Notes

### For SymbolCollector Tests

**Example spec structure**:
```crystal
require "spec"
require "./ast_fixtures"

require "../../src/compiler/frontend/ast"
require "../../src/compiler/semantic/context"
require "../../src/compiler/semantic/collectors/symbol_collector"
require "../../src/compiler/semantic/symbol"

describe CrystalV2::Compiler::Semantic::SymbolCollector do
  describe "def collection" do
    it "collects method definitions" do
      arena = CrystalV2::Compiler::Frontend::AstArena.new

      # Use fixtures to create AST
      def_id = AstFixtures.make_def(arena, "greet", params: ["name"])

      program = CrystalV2::Compiler::Frontend::Program.new(arena, [def_id])
      context = CrystalV2::Compiler::Semantic::Context.new(
        CrystalV2::Compiler::Semantic::SymbolTable.new
      )

      collector = CrystalV2::Compiler::Semantic::SymbolCollector.new(program, context)
      collector.collect

      # Verify MethodSymbol created
      symbol = context.symbol_table.lookup("greet")
      symbol.should_not be_nil
      symbol.should be_a(CrystalV2::Compiler::Semantic::MethodSymbol)

      method_sym = symbol.as(CrystalV2::Compiler::Semantic::MethodSymbol)
      method_sym.params.should eq(["name"])
      method_sym.scope.should_not be_nil  # Method has own scope
    end
  end

  describe "class collection" do
    it "collects class definitions with methods" do
      arena = CrystalV2::Compiler::Frontend::AstArena.new

      # Create: class Person
      #           def greet
      #           end
      #         end
      method_id = AstFixtures.make_def(arena, "greet")
      class_id = AstFixtures.make_class(arena, "Person", body: [method_id])

      program = CrystalV2::Compiler::Frontend::Program.new(arena, [class_id])
      context = CrystalV2::Compiler::Semantic::Context.new(
        CrystalV2::Compiler::Semantic::SymbolTable.new
      )

      collector = CrystalV2::Compiler::Semantic::SymbolCollector.new(program, context)
      collector.collect

      # Verify ClassSymbol created
      class_sym = context.symbol_table.lookup("Person")
      class_sym.should_not be_nil
      class_sym.should be_a(CrystalV2::Compiler::Semantic::ClassSymbol)

      # Verify method exists in class scope
      klass = class_sym.as(CrystalV2::Compiler::Semantic::ClassSymbol)
      method_in_class = klass.scope.lookup("greet")
      method_in_class.should_not be_nil
      method_in_class.should be_a(CrystalV2::Compiler::Semantic::MethodSymbol)
    end
  end
end
```

private def handle_def(node_id, node)
  name = String.new(node.def_name.not_nil!)
  params = node.def_params || [] of String

  # Create method scope
  method_scope = SymbolTable.new(@context.symbol_table)

  # Create symbol
  symbol = MethodSymbol.new(
    name, node_id,
    params: params,
    scope: method_scope
  )

  # Define in current scope
  if @context.symbol_table.local?(name)
    @context.symbol_table.redefine(name, symbol)
  else
    @context.symbol_table.define(name, symbol)
  end

private def handle_class(node_id, node)
  name = String.new(node.class_name.not_nil!)

  existing = @context.symbol_table.lookup(name)
  if existing && existing.is_a?(ClassSymbol)
