```crystal
private def visit(node_id : ExprId)
  case node.kind
  when .macro_def?
    handle_macro_def(node_id, node)
  when .def?
    handle_def(node_id, node)  # NEW
  when .class_def?

  end
end

private def handle_def(node_id, node)
  name = String.new(node.def_name.not_nil!)
  params = extract_params(node)  # Parse parameter list

  # Create method-local scope
  method_scope = SymbolTable.new(@context.symbol_table)

  symbol = MethodSymbol.new(
    name, node_id,
    params: params,
    scope: method_scope
  )

  @context.symbol_table.define(name, symbol)

  # TODO: Visit body with method_scope as current
end

private def handle_class_def(node_id, node)
  name = String.new(node.class_name.not_nil!)

  # Check for existing class (reopening)
  if existing = @context.symbol_table.lookup(name)
    if existing.is_a?(ClassSymbol)
      # Reopen existing class - use its scope
      class_scope = existing.scope
    else
      # Error: name conflict
      return
    end
  else
    # New class
    class_scope = SymbolTable.new  # No parent - class scope isolated
    symbol = ClassSymbol.new(name, node_id, scope: class_scope)
    @context.symbol_table.define(name, symbol)
  end

  # TODO: Visit class body with class_scope as current
end
```
